    function md(t){
      return t
        .replace(/&/g,'&amp;').replace(/</g,'&lt;')
        .replace(/^### (.*)$/gm,'<h4>$1</h4>')
        .replace(/^## (.*)$/gm,'<h4>$1</h4>')
        .replace(/\*\*(.+?)\*\*/g,'<b>$1</b>')
        .replace(/(?<!\*)\*(?!\*)(.+?)(?<!\*)\*(?!\*)/g,'<i>$1</i>')
        .replace(/^[-•] (.*)$/gm,'<li>$1</li>')
        .replace(/(<li>[\s\S]*?<\/li>)/g,'<ul>$1</ul>')
        .replace(/\n{2,}/g,'</p><p>')
        .replace(/\n/g,'<br>');
    }
